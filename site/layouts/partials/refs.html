{{- /* 参考文献の番号付きリストの各項目に id="ref-n" を付ける（FR1.4）。 引数: 本文の HTML。見出し「参考文献 {#references}」より後ろの
    だけを対象にし、 分類ごとに分かれたリスト（
      など）の番号を、そのまま id に使う。 */ -}} {{- $c := string . -}} {{- $marker := `

      ` -}} {{- if in $c $marker -}} {{- $parts := split $c $marker -}} {{- $tail := index $parts 1 -}} {{- range findRESubmatch `(?s)(.*?)

    ` $tail -}} {{- $start := 1 -}} {{- with index . 1 }}{{ $start = int . }}{{ end -}} {{- $items := after 1 (split (index . 2) "
  1. ") -}} {{- $new := "" -}} {{- range $i, $item := $items -}} {{- $new = printf `%s
  2. %s` $new (add $start $i) $item -}} {{- end -}} {{- $open := cond (eq $start 1) "
      " (printf `
        ` $start) -}} {{- $tail = replace $tail (index . 0) (printf "%s%s
      " $open $new) -}} {{- end -}} {{- $c = print (index $parts 0) $marker $tail -}} {{- end -}} {{- return ($c | safeHTML) -}}
