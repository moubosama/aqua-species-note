{{- /* 比較ページの「数字の幅」1つ（R-02）。引数: rows（本種の繁殖記録）、min・max（欄の名前）、 unit（単位）、page（種ページ）。 数字の幅の書き方（「5〜7日」「5–7 days」）と、空のときの文言は site/i18n/*.toml。両方の欄に数字がある記録だけを使い、出典番号を種ページへのリンクにする。 */ -}} {{- $lo := -1.0 -}} {{- $hi := -1.0 -}} {{- $refs := slice -}} {{- range $row := .rows -}} {{- if and (isset $row $.min) (isset $row $.max) -}} {{- $a := float (index $row $.min) -}} {{- $b := float (index $row $.max) -}} {{- if or (lt $lo 0.0) (lt $a $lo) }}{{ $lo = $a }}{{ end -}} {{- if gt $b $hi }}{{ $hi = $b }}{{ end -}} {{- /* refs から数字だけを取り出す。「5、6」や全角の空白でも止まらない（D3、FR4）。正しい書き方は frontmatter の確認が知らせる */ -}} {{- range findRE `\d+` $row.refs }}{{ $refs = $refs | append (int .) }}{{ end -}} {{- end -}} {{- end -}} {{- $out := T "no_record" -}} {{- if ge $hi 0.0 -}} {{- $nums := dict "lo" (printf "%g" $lo) "hi" (printf "%g" $hi) "unit" .unit -}} {{- $text := cond (eq $lo $hi) (T "range_one" $nums) (T "range_many" $nums) -}} {{- $cites := "" -}} {{- range sort (uniq $refs) -}} {{- $cites = printf `%s［%d］` $cites $.page.RelPermalink . (T "cite") . . -}} {{- end -}} {{- $out = print $text $cites -}} {{- end -}} {{- return ($out | safeHTML) -}}
